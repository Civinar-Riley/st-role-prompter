```javascript
    function cleanupStaleInstance() {
        // 双开保护：清理上一个实例（可能是旧版本）留下的面板和样式，避免同 ID 元素互相干扰
        const prev = host[INSTANCE_TOKEN_KEY];
        if (prev && prev !== myToken) {
            console.warn('[角色书生成器] 检测到另一个实例（可能是新旧版本同时启用），本实例已接管；请到脚本库停用旧版本。');
        }
        host[INSTANCE_TOKEN_KEY] = myToken;
        const staleDrawer = doc.getElementById(DRAWER_ID);
        if (staleDrawer) staleDrawer.remove();
        const staleStyle = doc.getElementById(STYLE_ID);
        if (staleStyle) staleStyle.remove();
    }

    function openDrawer() {
        if (host[INSTANCE_TOKEN_KEY] !== myToken) {
            toast('检测到另一个角色世界书生成器在运行（可能新旧版本同时启用），请到脚本库只保留一个', 'error');
            return;
        }
        // 同一次点击里刚被"点击外部"关掉时不立刻重开，让脚本按钮表现为开/关切换
        if (Date.now() - lastClosedAt < 400) return;
        buildDrawer();
        const drawer = doc.getElementById(DRAWER_ID);
        applyPopMode();
        drawer.classList.add('cwbg-open');
        applyTheme();
        // ... 渲染内容
    }
    function closeDrawer() {
        const drawer = doc.getElementById(DRAWER_ID);
        if (drawer) drawer.classList.remove('cwbg-open');
        lastClosedAt = Date.now();
    }
    const drawerOpen = () => {
        const drawer = doc.getElementById(DRAWER_ID);
        return !!(drawer && drawer.classList.contains('cwbg-open'));

```

**值得抄的点**：`lastClosedAt < 400ms` 的判断解决了"点击面板外部关闭 → 同一次点击冒泡到脚本按钮 → 又打开"的经典问题。

```javascript
function makeDraggable(el, handle) {
  let dragging = false, moved = false, sx = 0, sy = 0, ox = 0, oy = 0;
  handle.addEventListener('mousedown', e => {
    if (e.button !== 0) return;
    if (e.target.closest && e.target.closest('button,input')) return;
    const r = el.getBoundingClientRect();
    if (r.width > 0 && getComputedStyle(el).right !== 'auto') {
      el.style.left = r.left + 'px'; el.style.top = r.top + 'px';
      el.style.right = 'auto'; el.style.bottom = 'auto';
    }
    dragging = true; moved = false; sx = e.clientX; sy = e.clientY; ox = r.left; oy = r.top;
    e.preventDefault();
  });
  doc.addEventListener('mousemove', e => {
    if (!dragging) return;
    const dx = e.clientX - sx, dy = e.clientY - sy;
    if (!moved && Math.hypot(dx, dy) < 5) return;
    moved = true;
    el.style.left = Math.min(Math.max(ox + dx, 0), doc.documentElement.clientWidth - el.offsetWidth) + 'px';
    el.style.top = Math.min(Math.max(oy + dy, 0), doc.documentElement.clientHeight - el.offsetHeight) + 'px';
  });
  doc.addEventListener('mouseup', () => { dragging = false; });
}
```

要点：`pointerdown/move/up` + 3px 位移阈值区分点击与拖动；位置存 localStorage（`multi-gen-fab-pos`）；右键隐藏；`resize` 时夹取到边界内；`touch-action: none` 支持触屏。
