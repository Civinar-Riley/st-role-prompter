```javascript
    function cleanupStaleInstance() {
        // 双开保护：清理上一个实例（可能是旧版本）留下的面板和样式，避免同 ID 元素互相干扰
        const prev = host[INSTANCE_TOKEN_KEY];
        if (prev && prev !== myToken) {
            console.warn('[角色书生成器] 检测到另一个实例（可能是新旧版本同时启用），本实例已接管；请到脚本库停用旧版本。');
        }
        host[INSTANCE_TOKEN_KEY] = myToken;
        const staleDrawer = doc.getElementById(DRAWER_ID);
        if (staleDrawer) staleDrawer.remove();
        const staleStyle = doc.getElementById(STYLE_ID);
        if (staleStyle) staleStyle.remove();

    function openDrawer() {
        if (host[INSTANCE_TOKEN_KEY] !== myToken) {
            toast('检测到另一个角色世界书生成器在运行（可能新旧版本同时启用），请到脚本库只保留一个', 'error');
            return;
        }
        // 同一次点击里刚被"点击外部"关掉时不立刻重开，让脚本按钮表现为开/关切换
        if (Date.now() - lastClosedAt < 400) return;
        buildDrawer();
        const drawer = doc.getElementById(DRAWER_ID);
        applyPopMode();
        drawer.classList.add('cwbg-open');
        applyTheme();
        // ... 渲染内容
    }
    function closeDrawer() {
        const drawer = doc.getElementById(DRAWER_ID);
        if (drawer) drawer.classList.remove('cwbg-open');
        lastClosedAt = Date.now();
    }
    const drawerOpen = () => {
        const drawer = doc.getElementById(DRAWER_ID);
        return !!(drawer && drawer.classList.contains('cwbg-open'));

```

```javascript
    function detectDark() {
        // 酒馆没有 body.dark 之类的明暗类，改为计算背景色亮度；取不到时默认暗色
        try {
            const body = doc.body;
```

```javascript
function showModal(opts) {
```

### C4. `酒馆助手脚本-多功能生成器v4.7.2.json`（**第三方脚本**）

> ⚠️ 作者署名 **@清风携语至**，脚本内附**自定义授权协议**。
> **本节只记录思路，不摘录源码**——未经作者许可不宜转载。
> 本项目没有抄它的代码，下面这些是阅读后总结的通用做法。

**可拖拽悬浮球（FAB）的实现要点**：

- 用一个 `position: fixed` 的圆形 div，`z-index` 给到很高（如 999990）
- 拖动用 `pointerdown / pointermove / pointerup`，配合约 3px 的位移阈值区分「点击」与「拖动」
- 位置存进 localStorage，下次打开还在原位
- 右键（`contextmenu`）隐藏，隐藏前先 `preventDefault`
- 窗口 `resize` 时把球夹回视口内
- `touch-action: none` 让触屏也能拖
- 外观常用项：`border-radius: 50%`、半透明背景 + `backdrop-filter: blur()`、圆形投影

**启动骨架的要点**：

- 先加载配置，再创建悬浮球
- 注册脚本按钮前先 `typeof` 探测（老版本酒馆助手可能没有这些函数）
- 监听 `tavern_events.CHAT_CHANGED` 重载上下文
- 页面卸载时移除自己创建的 DOM

> 本项目 v1.0.0 的悬浮球就是按这些要点自己实现的，
> 代码见 [`05-UI模式.md`](05-UI模式.md) 第三节（那份是本项目自己的代码）。
