# 05 UI 模式

本项目的界面 = 一个入口（脚本按钮 + 可选悬浮球） + 一个管理面板（悬浮/居中） + 若干弹窗（增改条目、导入导出）。以下都是本地真实项目里验证过的代码骨架。

---

## 一、入口：脚本按钮（必做）

```javascript
const BUTTON_NAME = '角色名注入';

function registerScriptButton(onClick) {
  if (typeof appendInexistentScriptButtons === 'function'
      && typeof getButtonEvent === 'function'
      && typeof eventOn === 'function') {
    appendInexistentScriptButtons([{ name: BUTTON_NAME, visible: true }]);
    eventOn(getButtonEvent(BUTTON_NAME), onClick);
    return true;
  }
  console.warn('[角色名注入] 脚本按钮 API 不可用');
  return false;
}
```

按钮会出现在输入框上方（酒馆助手脚本按钮栏）。

## 二、入口降级：输入框旁的按钮

脚本按钮不可用时的兜底，真实项目 `NPC资料生成器` 的三级回退：

```javascript
let entry = doc.createElement('div');
entry.id = 'rni-entry';
entry.textContent = '🎭';
try {
  const rightForm = doc.querySelector('#rightSendForm');
  const sendBut = rightForm && rightForm.querySelector('#send_but');
  if (rightForm) {
    if (sendBut) rightForm.insertBefore(entry, sendBut);
    else rightForm.appendChild(entry);
  } else {
    const sheld = doc.querySelector('#form_sheld');
    if (sheld) sheld.appendChild(entry);
    else { entry.id = 'rni-entry-fallback'; doc.body.appendChild(entry); }
  }
  entry.addEventListener('click', togglePanel);
} catch (e) { console.error(e); }
```

选择器优先级：`#rightSendForm`（输入框右侧按钮区）→ `#form_sheld` → 页面右下角固定按钮。

**注意**：`#send_but` 等按钮初始带 `displayNone` 类、由 JS 按状态显隐，所以**不要依赖某个兄弟元素一定可见**——插到 `#send_but` 之前仍然是正确做法（该元素始终存在，只是可能隐藏）。另外输入框的父容器是 `#nonQRFormItems`，用户启用快速回复的"替换输入栏"功能时可能整个容器被换掉，届时 `#rightSendForm` 也会消失，所以保留最底层的兜底按钮。详见 [10-输入框机制源码核实.md](10-输入框机制源码核实.md) 第一、五节。

## 三、入口：可拖动悬浮球（推荐加）

参考 `酒馆助手脚本-多功能生成器v4.7.2` 的实现，用 pointer 事件 + 3px 位移阈值区分点击与拖动：

```javascript
const FAB_SIZE = 48;
const EDGE_GAP = 8;
const POS_KEY = 'rni_fab_pos';

function createFloatingBall(host, doc, onOpen) {
  const old = doc.getElementById('rni-fab');
  if (old) old.remove();

  const pos = loadLocal(POS_KEY, {
    x: host.innerWidth - FAB_SIZE - 16,
    y: host.innerHeight - 120,
  });

  const ball = doc.createElement('div');
  ball.id = 'rni-fab';
  ball.title = '角色名注入（点击打开，右键隐藏）';
  ball.textContent = '🎭';
  ball.style.cssText = `
    position:fixed; left:${pos.x}px; top:${pos.y}px;
    width:${FAB_SIZE}px; height:${FAB_SIZE}px; border-radius:50%;
    display:flex; align-items:center; justify-content:center;
    font-size:22px; color:#fff; cursor:grab; z-index:999990;
    background:rgba(15,23,42,.92); backdrop-filter:blur(10px);
    border:1px solid rgba(255,255,255,.2);
    box-shadow:0 4px 14px rgba(0,0,0,.4);
    user-select:none; touch-action:none;
  `;

  let start = {}, base = {}, moved = false;

  const onMove = (e) => {
    const dx = e.clientX - start.x, dy = e.clientY - start.y;
    if (!moved && Math.abs(dx) < 3 && Math.abs(dy) < 3) return;
    moved = true;
    const nx = Math.max(EDGE_GAP, Math.min(base.x + dx, host.innerWidth - FAB_SIZE - EDGE_GAP));
    const ny = Math.max(EDGE_GAP, Math.min(base.y + dy, host.innerHeight - FAB_SIZE - EDGE_GAP));
    ball.style.left = nx + 'px';
    ball.style.top = ny + 'px';
  };
  const onUp = () => {
    host.removeEventListener('pointermove', onMove);
    host.removeEventListener('pointerup', onUp);
    ball.style.cursor = 'grab';
    if (moved) {
      saveLocal(POS_KEY, { x: parseFloat(ball.style.left), y: parseFloat(ball.style.top) });
      setTimeout(() => { moved = false; }, 100);   // 防止拖动结束触发 click
    }
  };

  ball.addEventListener('pointerdown', (e) => {
    if (e.button !== 0) return;
    e.preventDefault();
    moved = false;
    start = { x: e.clientX, y: e.clientY };
    const r = ball.getBoundingClientRect();
    base = { x: r.left, y: r.top };
    ball.style.cursor = 'grabbing';
    host.addEventListener('pointermove', onMove);
    host.addEventListener('pointerup', onUp);
  });

  ball.addEventListener('click', (e) => {
    if (moved) return;
    e.stopPropagation();
    onOpen();
  });

  ball.addEventListener('contextmenu', (e) => {
    e.preventDefault();
    if (host.confirm('隐藏悬浮球？可在面板设置里重新开启。')) {
      saveLocal('rni_show_fab', false);
      ball.remove();
    }
  });

  host.addEventListener('resize', () => {
    const nx = Math.max(EDGE_GAP, Math.min(parseFloat(ball.style.left), host.innerWidth - FAB_SIZE - EDGE_GAP));
    const ny = Math.max(EDGE_GAP, Math.min(parseFloat(ball.style.top), host.innerHeight - FAB_SIZE - EDGE_GAP));
    ball.style.left = nx + 'px';
    ball.style.top = ny + 'px';
  });

  doc.body.appendChild(ball);
}
```

关键点：`touch-action: none` 让手机也能拖；`resize` 时把球夹回视口内；位置存 localStorage。

```javascript
function openPanel() {

  doc.getElementById('rni-panel')?.classList.remove('open');

  const p = doc.getElementById('rni-panel');
  if (p && p.classList.contains('open')) closePanel(); else openPanel();
```

## 五、弹窗：原生 Promise 化

增改条目 / 确认删除 / 导入导出都用这个（合并了两个真实项目的写法）：

```javascript
let openModalEl = null;

function showModal({ title, contentEl, buttons }) {
  closeModal();
  const mask = doc.createElement('div');
  mask.className = 'rni-modal-mask';
  mask.style.cssText = 'position:fixed;inset:0;z-index:99999;background:rgba(0,0,0,.5);display:flex;align-items:center;justify-content:center;';

  const box = doc.createElement('div');
  box.className = 'rni-modal';
  box.style.cssText = 'min-width:min(460px,90vw);max-width:min(720px,92vw);max-height:86vh;overflow:auto;background:var(--rni-bg);color:var(--rni-fg);border:1px solid var(--rni-border);border-radius:10px;padding:16px 18px;box-shadow:0 12px 40px rgba(0,0,0,.5);';

  const h = doc.createElement('h3');
  h.textContent = title;
  h.style.cssText = 'margin:0 0 12px;font-size:16px;';
  box.appendChild(h);

  if (contentEl) box.appendChild(contentEl);

  const actions = doc.createElement('div');
  actions.style.cssText = 'display:flex;justify-content:flex-end;gap:8px;margin-top:16px;';
  for (const b of buttons || []) {
    const btn = doc.createElement('button');
    btn.textContent = b.text;
    btn.style.cssText = 'padding:6px 14px;border-radius:6px;cursor:pointer;border:1px solid var(--rni-border);background:var(--rni-panel);color:var(--rni-fg);';
    if (b.primary) btn.style.cssText += 'background:var(--rni-accent);border-color:transparent;color:#fff;';
    if (b.danger)  btn.style.cssText += 'background:var(--rni-danger);border-color:transparent;color:#fff;';
    btn.onclick = async () => {
      let ok = true;
      if (b.onClick) ok = (await b.onClick()) !== false;
      if (ok !== false) closeModal();
    };
    actions.appendChild(btn);
  }
  box.appendChild(actions);

  mask.appendChild(box);
  mask.addEventListener('mousedown', (e) => { if (e.target === mask) closeModal(); });
  doc.body.appendChild(mask);
  openModalEl = mask;
  return mask;
}

function closeModal() {
  if (openModalEl) { openModalEl.remove(); openModalEl = null; }
}

function modalOpen() { return !!openModalEl; }
```

要点：`onClick` 返回 `false` 可以阻止关闭（用于表单校验失败时保留弹窗）；点击遮罩关闭；弹窗 z-index 必须高于面板。

## 六、拖动（面板 / 弹窗）

**不要用 Pointer Events 做面板拖动**（部分 WebView 不支持），用 mouse + touch 双事件：

```javascript
function makeDraggable(el, handle) {
  let dragging = false, moved = false, sx = 0, sy = 0, ox = 0, oy = 0;

  const start = (x, y) => {
    const r = el.getBoundingClientRect();
    if (r.width > 0 && host.getComputedStyle(el).right !== 'auto') {
      el.style.left = r.left + 'px';
      el.style.top = r.top + 'px';
      el.style.right = 'auto';
      el.style.bottom = 'auto';
    }
    dragging = true; moved = false; sx = x; sy = y; ox = r.left; oy = r.top;
  };
  const move = (x, y) => {
    if (!dragging) return;
    const dx = x - sx, dy = y - sy;
    if (!moved && Math.hypot(dx, dy) < 5) return;
    moved = true;
    el.style.left = Math.min(Math.max(ox + dx, 0), doc.documentElement.clientWidth - el.offsetWidth) + 'px';
    el.style.top  = Math.min(Math.max(oy + dy, 0), doc.documentElement.clientHeight - el.offsetHeight) + 'px';
  };
  const end = () => { dragging = false; };

  handle.addEventListener('mousedown', (e) => {
    if (e.button !== 0) return;
    if (e.target.closest && e.target.closest('button,input,textarea,select')) return;
    e.preventDefault(); start(e.clientX, e.clientY);
  });
  doc.addEventListener('mousemove', (e) => move(e.clientX, e.clientY));
  doc.addEventListener('mouseup', end);

  handle.addEventListener('touchstart', (e) => {
    if (e.target.closest && e.target.closest('button,input,textarea,select')) return;
    const t = e.touches[0]; start(t.clientX, t.clientY);
  }, { passive: true });
  doc.addEventListener('touchmove', (e) => {
    const t = e.touches[0]; move(t.clientX, t.clientY);
  }, { passive: true });
  doc.addEventListener('touchend', end);
}
```

拖动结束后可能紧跟一个 click，参考项目用 `stopImmediatePropagation` 或时间戳拦掉，避免"拖完松手就打开/关闭"。

常用变量：`--SmartThemeBlurTintColor`、`--SmartThemeBorderColor`、`--SmartThemeBodyColor`、`--SmartThemeQuoteColor`、`--SmartThemeEmColor`。

### 方案 B：自算明暗（面板用自定义配色时）

酒馆没有 `body.dark` 这类类名，真实项目改用背景色亮度判断：

```javascript
function detectDark() {
  try {
    if (doc.body.classList.contains('dark') || doc.documentElement.classList.contains('dark')) return true;
    let el = doc.body;
    for (let i = 0; i < 4 && el; i++) {
      const bg = host.getComputedStyle(el).backgroundColor;
      const m = bg && bg.match(/rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\s*(?:,\s*([\d.]+))?\)/);
      if (m && (m[4] === undefined || parseFloat(m[4]) > 0.5)) {
        const lum = (0.2126 * +m[1] + 0.7152 * +m[2] + 0.0722 * +m[3]) / 255;
        return lum < 0.55;
      }
      el = el.parentElement || doc.documentElement;
    }
  } catch (e) {}
  return true;
}
```

给面板根元素加 `.dark` / `.light` 类，CSS 里两套变量。

```javascript
toastr.success('已保存');
```
